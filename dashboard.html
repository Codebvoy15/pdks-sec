async function loadAvailableContextsOnStartup() { try { // Updated to use the proxy-hardened context route const response = await fetch('/pdks/api/contexts'); if (!response.ok) throw new Error("CONTEXT_API_UNREACHABLE"); activeClusterContexts = await response.json(); const selector = document.getElementById('globalClusterSelector'); activeClusterContexts.sort((a,b) => a.Name.localeCompare(b.Name)).forEach(ctx => { let opt = document.createElement('option'); opt.value = ctx.context; opt.textContent = `☸ ${ctx.Name.toUpperCase()}`; selector.appendChild(opt); }); } catch (err) { showStatusBanner("Fatal Failure loading target contexts from Kubeconfig profiles.", "error"); } } async function handleClusterSelection() { const targetCtx = document.getElementById('globalClusterSelector').value; const banner = document.getElementById('clusterStatusBanner'); if (!targetCtx) { showStatusBanner("◀ Please select a valid target context option from the global cluster dropdown menu above to begin dynamic live loading.", "info"); clearDashboardDisplays(); return; } banner.className = "status-banner status-info"; banner.innerHTML = `
Initializing authenticated client-go pipeline connection parameters for context [${targetCtx}]...
`; banner.style.display = "block"; try { // Updated query target path const response = await fetch(`/pdks/api/query?context=${encodeURIComponent(targetCtx)}`); const payload = await response.json(); if (payload.error) { showStatusBanner(`❌ Context Connection Aborted: ${payload.error}`, "error"); clearDashboardDisplays(); return; } banner.style.display = "none"; renderDashboardDataMetrics(payload); } catch (err) { showStatusBanner(" Core connection fault reached during target network loop scan.", "error"); clearDashboardDisplays(); } }
